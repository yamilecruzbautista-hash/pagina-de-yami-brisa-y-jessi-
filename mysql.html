Módulo: MySQL (Base de Datos, Tablas, Registros y Consultas)

Crear Base de Datos y Tablas:

CREATE DATABASE escuela;
USE escuela;

CREATE TABLE alumnos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50),
    edad INT
);

Insertar Registros:

INSERT INTO alumnos (nombre, edad) VALUES ('Carlos', 17), ('Ana', 16);

Consultas Básicas (SELECT):

SELECT * FROM alumnos;

Modificar / Actualizar Registros (UPDATE):

UPDATE alumnos SET edad = 18 WHERE id = 1;

Eliminar Registros (DELETE):

DELETE FROM alumnos WHERE id = 2;
